Program 1: Accept a student's name, roll number and branch, then display the details

Open C compiler
/* Question 1: Accept a student's name, roll number and branch, then display the details */
#include <stdio.h>

int main()
{
    char name[50], branch[30];
    int roll;
    printf("Enter student name: ");
    scanf(" %[^\n]", name);
    printf("Enter roll number: ");
    scanf("%d", &roll);
    printf("Enter branch: ");
    scanf(" %[^\n]", branch);
    printf("\n--- Student Details ---\n");
    printf("Name   : %s\n", name);
    printf("Roll No: %d\n", roll);
    printf("Branch : %s\n", branch);
    return 0;
}

Program 2: Accept two integers and display their sum, difference, product and quotient

Open C compiler
/* Question 2: Accept two integers and display their sum, difference, product and quotient */
#include <stdio.h>

int main()
{
    int a, b;
    printf("Enter two integers: ");
    scanf("%d %d", &a, &b);
    printf("Sum        = %d\n", a + b);
    printf("Difference = %d\n", a - b);
    printf("Product    = %d\n", a * b);
    if (b != 0)
        printf("Quotient   = %d\n", a / b);
    else
        printf("Quotient   = Cannot divide by zero\n");
    return 0;
}

Program 4: Accept the length and breadth of a rectangle and calculate its area

{
    float length, breadth, area;

Program 5: Accept the radius of a circle and calculate its area

Open C compiler
/* Question 5: Accept the radius of a circle and calculate its area */
#include <stdio.h>
#define PI 3.14159

int main()
{
    float r, area;
    printf("Enter radius: ");
    scanf("%f", &r);
    area = PI * r * r;                /* area = pi * r * r */
    printf("Area of circle = %.2f\n", area);
    return 0;
}

Program 8: Accept five subject marks and calculate the total marks and average

Open C compiler
/* Question 8: Accept five subject marks and calculate the total marks and average */
#include <stdio.h>

int main()
{
    float m1, m2, m3, m4, m5, total, average;
    printf("Enter marks of 5 subjects: ");
    scanf("%f %f %f %f %f", &m1, &m2, &m3, &m4, &m5);
    total = m1 + m2 + m3 + m4 + m5;
    average = total / 5;
    printf("Total marks = %.2f\n", total);
    printf("Average     = %.2f\n", average);
    return 0;
}
